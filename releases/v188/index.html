📖 使用方法

基本的な使用（変更なし）

from dictsqlite import DictSQLite

# 通常の使用
db = DictSQLite('example.db')
db['key'] = 'value'

新しいJSON モード（実験的）

# JSON モードの使用（実験的）
db = DictSQLite('data.db', json_mode=True)
db['complex_data'] = {
    'nested': {
        'structure': ['array', 'data'],
        'timestamp': '2025-09-20'
    }
}

スキーマ指定の正しい使用

# 型注釈修正後の正しい使用
db = DictSQLite('schema_db.db', schema='custom_schema.sql')
# または
db = DictSQLite('auto_db.db', schema=None)  # デフォルト
