🌐 多言語サポート

from dictsqlite import DictSQLite

# English documentation example
from dictsqlite import DictSQLite

# Create database

📋 AI生成コード例

# AI が生成した高度な使用例
from dictsqlite import DictSQLite, expiring_dict

# 期限付きキャッシュシステム
cache = expiring_dict(3600)  # 1時間
db = DictSQLite('cache.db')

# 複合的なデータ管理
users = db.table('users')
sessions = db.table('sessions')

with db.transaction():
    users['alice'] = {'role': 'admin', 'created': '2025-09-18'}
    sessions['alice_session'] = {'expires': '2025-09-19'}
